    pure function fmt_g(x) result(s)

    real(wp), intent(in) :: x !! the number
    character(len=:), allocatable :: s

    character(len=32) :: buf
    integer :: ios

    write(buf, '(ES16.9)', iostat=ios) x
    if (ios /= 0) then
        s = '****'
    else
        s = trim(adjustl(buf))
    end if

    end function fmt_g

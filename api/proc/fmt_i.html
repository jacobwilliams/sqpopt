    pure function fmt_i(i) result(s)

    integer, intent(in) :: i !! the number
    character(len=:), allocatable :: s

    character(len=32) :: buf
    integer :: ios

    write(buf, '(I0)', iostat=ios) i
    if (ios /= 0) then
        s = '****'
    else
        s = trim(adjustl(buf))
    end if

    end function fmt_i

    pure function fmt_e(x) result(s)

    real(wp), intent(in) :: x !! the number
    character(len=:), allocatable :: s

    character(len=32) :: buf
    integer :: ios

    write(buf, '(ES10.3)', iostat=ios) x
    if (ios /= 0) then
        s = '****'
    else
        s = trim(adjustl(buf))
    end if

    end function fmt_e

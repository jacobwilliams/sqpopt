    function wall_time() result(t)

    use, intrinsic :: iso_fortran_env, only: int64

    real(wp) :: t

    integer(int64) :: count, rate

    call system_clock(count, rate)
    t = real(count, wp)/real(rate, wp)

    end function wall_time

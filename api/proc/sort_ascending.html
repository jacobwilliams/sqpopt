private pure subroutine sort_ascending(v, w)

sort the eigenvalues into ascending order, with their eigenvectors (selection sort: the orders here are small).

    pure subroutine sort_ascending(v, w)

    real(wp), dimension(:,:), intent(inout) :: v !! the eigenvectors (columns) `dimension(n,n)`
    real(wp), dimension(:),   intent(inout) :: w !! the eigenvalues `dimension(n)`

    integer  :: i, j, r
    real(wp) :: t

    do i = 1, size(w) - 1
        j = minloc(w(i:), dim=1) + i - 1
        if (j == i) cycle
        t = w(i); w(i) = w(j); w(j) = t
        do r = 1, size(v,1)
            t = v(r,i); v(r,i) = v(r,j); v(r,j) = t
        end do
    end do

    end subroutine sort_ascending

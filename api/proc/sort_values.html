    pure subroutine sort_values(w)

    real(wp), dimension(:), intent(inout) :: w !! the eigenvalues `dimension(n)`

    integer  :: i, j
    real(wp) :: t

    do i = 1, size(w) - 1
        j = minloc(w(i:), dim=1) + i - 1
        if (j == i) cycle
        t = w(i); w(i) = w(j); w(j) = t
    end do

    end subroutine sort_values

merit_slack

public pure function merit_slack(phi0) result(slack)

the roundoff-level slack allowed when comparing a trial merit function (or objective) value against the current one, (as in IPOPT's Compare_le).

    pure function merit_slack(phi0) result(slack)

    real(wp), intent(in) :: phi0 !! merit function (or objective) value at the current point
    real(wp) :: slack

    slack = 10.0_wp*epsilon(1.0_wp)*max(1.0_wp, abs(phi0))

    end function merit_slack

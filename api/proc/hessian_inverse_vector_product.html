hessian_inverse_vector_product

private subroutine hessian_inverse_vector_product(me, v, d)

compute the matrix-free inverse-Hessian-vector product . For the BFGS mode this uses the classic two-loop recursion. For the SR1 mode (no simple closed-form inverse exists) this uses a small internal conjugate-gradient solve driven by hessian_vector_product.

    subroutine hessian_inverse_vector_product(me, v, d)

    class(sqpopt_hessian_type), intent(inout) :: me
    real(wp), dimension(:), intent(in)  :: v  !! input vector `dimension(n)`
    real(wp), dimension(:), intent(out) :: d  !! result `dimension(n)`

    real(wp), dimension(:), allocatable :: alpha !! temporary storage for the two-loop recursion coefficients in L-BFGS
    real(wp), dimension(:), allocatable :: q !! temporary vector used in the two-loop recursion
    integer :: i !! loop index for the two-loop recursion
    integer :: c !! column of the `i`-th pair
    real(wp) :: beta

    if (me%use_sr1 .or. me%exact) then
        call hessian_cg_solve(me, v, d)
        return
    end if

    ! standard L-BFGS two-loop recursion:
    allocate(alpha(me%n_history), q(me%n))
    q = v
    do i = me%n_history, 1, -1
        c = pair_col(me, i)
        alpha(i) = me%rho(c)*dot_product(me%s(:,c), q)
        q = q - alpha(i)*me%y(:,c)
    end do

    d = me%gamma*q

    do i = 1, me%n_history
        c = pair_col(me, i)
        beta = me%rho(c)*dot_product(me%y(:,c), d)
        d = d + me%s(:,c)*(alpha(i) - beta)
    end do

    end subroutine hessian_inverse_vector_product

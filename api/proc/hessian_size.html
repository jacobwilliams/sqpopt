    pure function hessian_size(me) result(h)

    class(sqpopt_hessian_type), intent(in) :: me
    real(wp) :: h

    h = 1.0_wp
    if (me%exact) then
        if (size(me%h_val) > 0) h = max(1.0_wp, maxval(abs(me%h_val)))
    else
        h = max(1.0_wp, 1.0_wp/me%gamma)
    end if

    end function hessian_size

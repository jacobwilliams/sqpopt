symmetric_solver_destroy

private subroutine symmetric_solver_destroy(me)

free the solver, and return it to its initial state. It does nothing to a solver that was never initialized.

    subroutine symmetric_solver_destroy(me)

    class(sqpopt_symmetric_solver_type), intent(inout) :: me

    if (allocated(me%ldl)) then
        call me%ldl%free()
        deallocate(me%ldl)
    end if
    me%ready         = .false.
    me%factored      = .false.
    me%backend       = 0
    me%n             = 0
    me%n_negative    = 0
    me%n_null        = 0
    me%inertia_known = .true.
    me%n_factor      = 0
    me%n_solve       = 0
    me%time          = 0.0_wp
    me%out_of_memory = .false.

    end subroutine symmetric_solver_destroy

dense_lu_factor

public pure subroutine dense_lu_factor(a, piv, ok)

in-place LU factorization with partial pivoting of a small dense matrix a (e.g. the middle matrix of sqpopt_hessian_module, whose order is 2*max_history at most, independent of n). ok is false if a pivot is negligible relative to the matrix's largest element.

    pure subroutine dense_lu_factor(a, piv, ok)

    real(wp), dimension(:,:), intent(inout) :: a   !! matrix, overwritten by its `L` (unit, below the diagonal) and `U` factors
    integer,  dimension(:),   intent(out)   :: piv !! `piv(p)` is the row swapped with row `p` at step `p`
    logical,                  intent(out)   :: ok  !! false if `a` is (numerically) singular

    integer :: i, p, k2
    real(wp) :: amax, tol

    k2 = size(a,1)
    ok = .true.
    tol = 1.0e-14_wp*max(maxval(abs(a)), tiny(1.0_wp))
    do p = 1, k2
        piv(p) = p - 1 + maxloc(abs(a(p:k2,p)), dim=1)
        amax = abs(a(piv(p),p))
        if (amax <= tol) then
            ok = .false.
            return
        end if
        if (piv(p) /= p) a([p,piv(p)],:) = a([piv(p),p],:)
        do i = p+1, k2
            a(i,p) = a(i,p)/a(p,p)
            a(i,p+1:k2) = a(i,p+1:k2) - a(i,p)*a(p,p+1:k2)
        end do
    end do

    end subroutine dense_lu_factor

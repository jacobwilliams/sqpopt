    pure real(wp) function hessian_low_rank_scale(me) result(scale)

    class(sqpopt_hessian_type), intent(in) :: me

    integer  :: i, c
    real(wp) :: theta

    theta = 1.0_wp/me%gamma
    scale = 0.0_wp
    do i = 1, me%n_history
        c = pair_col(me, i)
        scale = max(scale, me%yy(c,c) + theta**2*me%ss(c,c))
    end do

    end function hessian_low_rank_scale
